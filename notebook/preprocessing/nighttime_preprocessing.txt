import arcpy
from arcpy.sa import *
import os

arcpy.env.overwriteOutput = True


# ---------------------------
# USER PARAMETERS
inRaster = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20221225\Himawari_UTM_C.tif"
inPolygon = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\shapefile\ppb_square_b.shp"
outRaster = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20221225\00_Himawari_idw_1000m.tif"
cellSize = 1000            # target cell size in map units (1000 m)
maxDistance = 50000        # max extrapolation distance (meters) — adjust as needed
power = 2.0                # IDW power parameter (2 is typical)
# ---------------------------

try:
    arcpy.CheckOutExtension("Spatial")

    # Ensure inputs exist
    if not arcpy.Exists(inRaster):
        raise Exception("Input raster not found: " + inRaster)
    if not arcpy.Exists(inPolygon):
        raise Exception("Input polygon not found: " + inPolygon)

    # Work in the scratch geodatabase
    scratchgdb = arcpy.env.scratchGDB
    if not scratchgdb:
        # fallback to in_memory if no scratchGDB
        scratchgdb = "in_memory"

    # 1) Make sure polygon is in same CRS as raster (project if needed)
    rasDesc = arcpy.Describe(inRaster)
    rasSR = rasDesc.spatialReference
    polyDesc = arcpy.Describe(inPolygon)
    polySR = polyDesc.spatialReference

    projectedPoly = inPolygon
    if rasSR.name != polySR.name:
        projectedPoly = os.path.join(scratchgdb, "poly_proj")
        arcpy.management.Project(inPolygon, projectedPoly, rasSR)
        arcpy.AddMessage("Projected polygon to raster CRS: {}".format(rasSR.name))
    else:
        arcpy.AddMessage("Polygon already in raster CRS: {}".format(rasSR.name))

    # 2) Convert raster to points (points only where raster had data)
    pts_fc = os.path.join(scratchgdb, "raster_points")
    if arcpy.Exists(pts_fc):
        arcpy.management.Delete(pts_fc)
    arcpy.conversion.RasterToPoint(inRaster, pts_fc, "VALUE")
    arcpy.AddMessage("RasterToPoint produced {} points".format(arcpy.GetCount_management(pts_fc)))

        # 3) Set Z field explicitly
    z_field = "grid_code"

    arcpy.AddMessage("Using '{}' as Z field for interpolation".format(z_field))

    # 4) Set environment to target area & cell size
    arcpy.env.snapRaster = inRaster     # keeps alignment if you want it matched to input raster
    arcpy.env.cellSize = cellSize
    arcpy.env.extent = projectedPoly    # limit processing extent to polygon (speeds up)

    # 5) Run IDW
    idw_raster = Idw(pts_fc, z_field, cellSize, power)

    # Save temporary raster and create Raster object for masking
    tmp_idw_path = r"in_memory\tmp_idw"
    idw_raster.save(tmp_idw_path)
    tmp_idw = Raster(tmp_idw_path)
    idw_raster.save(tmp_idw_path)
    tmp_idw = Raster(tmp_idw_path)  # <-- THIS is needed
    arcpy.AddMessage("IDW raster created (temporary).")

    # 6) Limit extrapolation by distance from known data
    dist_raster = EucDistance(pts_fc, maxDistance, cellSize)
    idw_masked = Con(Raster(dist_raster) <= maxDistance, tmp_idw)
    tmp_masked = os.path.join(scratchgdb, "tmp_idw_masked")
    idw_masked.save(tmp_masked)
    arcpy.AddMessage("Applied maxDistance mask ({} m).".format(maxDistance))

    # 7) Clip / mask to the polygon exactly
    final = ExtractByMask(tmp_masked, projectedPoly)
    # Save as 32-bit float
    final.save(outRaster)
    arcpy.AddMessage("Final extrapolated raster saved: {}".format(outRaster))

except Exception as e:
    arcpy.AddError("Error: {}".format(e))
    raise
finally:
    try:
        arcpy.CheckInExtension("Spatial")
    except:
        pass

import arcpy
from arcpy.sa import *

arcpy.CheckOutExtension("Spatial")
arcpy.env.overwriteOutput = True

# Input raster
raster_in = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20221225\03_modis_extract.tif"
filled_raster = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20221225\04_modis_fill.tif"

# Load raster
filled_raster_obj = Raster(raster_in)

iteration = 0
max_iterations = 10  # safety stop so it won't run forever

while True:
    iteration += 1
    
    # Smooth/estimate with neighborhood mean
    filled = FocalStatistics(filled_raster_obj, NbrRectangle(3, 3, "CELL"), "MEAN")
    
    # Fill only NoData cells
    new_filled_raster = Con(IsNull(filled_raster_obj), filled, filled_raster_obj)
    
    # Check if any NoData remains
    remaining_nodata = arcpy.GetRasterProperties_management(new_filled_raster, "ANYNODATA")
    remaining_nodata = remaining_nodata.getOutput(0)  # "1" = some NoData, "0" = none
    
    print(f"Iteration {iteration} complete. ANYNODATA = {remaining_nodata}")
    
    # Assign for next loop
    filled_raster_obj = new_filled_raster
    
    # Stop if no NoData remains or safety limit reached
    if remaining_nodata == "0" or iteration >= max_iterations:
        break

# Save result
filled_raster_obj.save(filled_raster)
print("✅ Filled raster saved:", filled_raster)

arcpy.CheckInExtension("Spatial")

### 4. MODIS Fill ###
#####################
import arcpy
from arcpy.sa import *

arcpy.CheckOutExtension("Spatial")
arcpy.env.overwriteOutput = True

# Input raster
raster_in = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20211212\03_modis_extract.tif"
filled_raster = r"C:\Users\percy\Documents\studies\visual_studio_code\PhD-chapter2-data-fusion-full-diurnal-lst\raw\night\20211212\04_modis_fill.tif"

# Load raster
filled_raster_obj = Raster(raster_in)

iteration = 0
max_iterations = 10  # safety stop so it won't run forever

while True:
    iteration += 1
    
    # Smooth/estimate with neighborhood mean
    filled = FocalStatistics(filled_raster_obj, NbrRectangle(3, 3, "CELL"), "MEAN")
    
    # Fill only NoData cells
    new_filled_raster = Con(IsNull(filled_raster_obj), filled, filled_raster_obj)
    
    # Check if any NoData remains
    remaining_nodata = arcpy.GetRasterProperties_management(new_filled_raster, "ANYNODATA")
    remaining_nodata = remaining_nodata.getOutput(0)  # "1" = some NoData, "0" = none
    
    print(f"Iteration {iteration} complete. ANYNODATA = {remaining_nodata}")
    
    # Assign for next loop
    filled_raster_obj = new_filled_raster
    
    # Stop if no NoData remains or safety limit reached
    if remaining_nodata == "0" or iteration >= max_iterations:
        break

# Save result
filled_raster_obj.save(filled_raster)
print("✅ Filled raster saved:", filled_raster)

arcpy.CheckInExtension("Spatial")

    mosaic_colormap_mode="FIRST"
)
